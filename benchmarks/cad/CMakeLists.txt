cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(FruitRenderBenchmark LANGUAGES CXX)
  if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Benchmark configurations" FORCE)
  endif()
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  endif()
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Render benchmarks require Release")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CORE REQUIRED IMPORTED_TARGET datamunge)
  set(fruit_render_core PkgConfig::CORE)
else()
  set(fruit_render_core datamunge::datamunge)
endif()
add_executable(fruit_render_bench fruit_render.cpp)
target_link_libraries(fruit_render_bench PRIVATE ${fruit_render_core})
target_compile_features(fruit_render_bench PRIVATE cxx_std_23)

add_executable(fruit_render_gpu_bench fruit_render_gpu.cpp)
target_link_libraries(fruit_render_gpu_bench PRIVATE ${fruit_render_core})
target_compile_features(fruit_render_gpu_bench PRIVATE cxx_std_23)

add_executable(fruit_wrist_bench fruit_wrist.cpp)
target_link_libraries(fruit_wrist_bench PRIVATE ${fruit_render_core})
target_compile_features(fruit_wrist_bench PRIVATE cxx_std_23)
