    class datamunge::rl::environments

    VsimTaskEnv

    class VsimTaskEnv : public datamunge::rl::environments::EnvBase
    #include <datamunge/rl/environments/vsim_tasks.hpp> source

    Own a vsim scene and expose its articulated task as a Datamunge environment. Enable DATAMUNGE_ENABLE_VSIM_TASKS before including dm/dm.hpp and link against the vsim SDK. Each reset restores the authored state before perturbing the bodies and joints with the environment RNG.

    Example
    namespace e=dm::rl::environments;
e::VsimTaskEnv task(e::VsimTask::Hopper,"hopper.json");
const auto obs=task.reset(3);
const auto step=task.step(e::Action({0.f,0.f,0.f}));
