    function datamunge::causal

    binary_ate

    #include <datamunge/causal/effects.hpp>
    #source
    BinaryATE binary_ate(
    const std::vector< double > & outcome,
    const std::vector< int > & treatment,
    const std::vector< double > & propensity,
    const std::vector< double > & control_mean,
    const std::vector< double > & treated_mean)

    Estimates an average treatment effect from supplied nuisance predictions.

    Parameters
    outcome

    Finite observed outcomes.

    treatment

    Treatment indicators (0 or 1), with both arms represented.

    propensity

    Treatment probabilities strictly between zero and one.

    control_mean

    Predicted outcomes under control, for all rows.

    treated_mean

    Predicted outcomes under treatment, for all rows.

    Returns

    Raw and adjusted point estimates and propensity extrema.

    Throws
    std::invalid_argument

    Invalid sizes, indicators or values.

    Example
    const auto fit = dm::causal::binary_ate({1, 3}, {0, 1}, {.5, .5}, {1, 1}, {3, 3});
