    template<typename Frame, typename Func, std::enable_if_t< std::conjunction_v< std::is_same< std::remove_cv_t< std::remove_reference_t< Frame > >, DataFrame >, std::is_invocable< Func, const DataFrame & > >, int > = 0>
decltype(auto) operator|(Frame && frame, Func && func)

    frame | fn: any callable taking the DataFrame is a pipeline step.

    Generic lambdas ([](const auto& df) { ... }) work: the overload first checks that the left side is a DataFrame, and only then whether func accepts it, so the lambda's body is never instantiated with a GroupedDataFrame.

    const auto top = dm::datasets::iris() | [](const auto& df) { return dm::head(df, 3); };
std::print("{}\n", top.nrows());   // 3
