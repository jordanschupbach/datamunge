file(GLOB_RECURSE CXX_SOURCES "${PROJECT_BINARY_DIR}/*.cpp")

  set(NEW_ENTRY "${PROJECT_BINARY_DIR}/datamungejs/src/${ENTRY}")
    list(APPEND NEW_SOURCES "${NEW_ENTRY}")
endforeach()
message(STATUS "Transformed SOURCES: ${NEW_SOURCES}")

# NOTE: I don't think this is necessary for datamungejs
# Copy source files into working directory and transform includes from system to local
foreach(filepath ${DATAMUNGE_SOURCES})
  file(READ ${filepath} input_content)
  string(REGEX REPLACE "#include<datamunge/([^>]*)>" "#include \"datamunge/\\1\"" output_content "${input_content}")
  string(REGEX REPLACE "#include <datamunge/([^>]*)>" "#include \"datamunge/\\1\"" output_content "${input_content}")
  file(WRITE ${filepath} "${output_content}")
endforeach()
