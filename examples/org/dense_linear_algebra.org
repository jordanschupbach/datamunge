This example compiles and runs a small C++ program against the local
=datamunge= headers. The batch exporter in [[file:../../init.el][init.el]]
loads the repo's =direnv= environment first, so the compiler and other project
tools are available to the source block.

#+begin_src shell
mkdir -p build/org
cat > build/org/dense_linear_algebra.cpp <<'CPP'
#include <datamunge/linalg/linalg.hpp>

#include <iostream>

using datamunge::linalg::DenseMatrix;

int main() {
    DenseMatrix<double> a(2, 2, {1, 2, 3, 4});
    DenseMatrix<double> b(2, 2, {5, 6, 7, 8});

    auto sum = a + b;
    auto product = a * b;

    std::cout << "sum(0,0) = " << sum(0, 0) << "\n";
    std::cout << "sum(1,1) = " << sum(1, 1) << "\n";
    std::cout << "product(0,0) = " << product(0, 0) << "\n";
    std::cout << "product(1,1) = " << product(1, 1) << "\n";
    std::cout << "frobenius(a) = " << a.norm_frobenius() << "\n";
    return 0;
}
CPP

c++ -std=c++17 -Iinclude build/org/dense_linear_algebra.cpp \
  -o build/org/dense_linear_algebra

./build/org/dense_linear_algebra
#+end_src

This second block fits a nonlinear regression curve using penalized cubic
B-splines (P-splines). The synthetic response is sampled from
=f(x) = 20 / sin(x + 0.15)= with additive Gaussian noise, and the fit uses many
basis functions plus a second-difference roughness penalty to stabilize the
curve.

#+begin_src shell
mkdir -p build/org
cat > build/org/bspline_regression.cpp <<'CPP'
#include <datamunge/fda/pspline.hpp>
#include <datamunge/plot/plot.hpp>
#include <datamunge/random/random.hpp>

#include <cmath>
#include <iomanip>
#include <iostream>
#include <vector>

using datamunge::fda::PSplineOptions;
using datamunge::fda::fit_pspline;
using datamunge::random::SplitMix64;
using namespace datamunge::plot;

int main() {
    constexpr std::size_t n = 500;
    constexpr double xmin = 0.0;
    constexpr double xmax = 1.0;
    constexpr double noise_sigma = 0.25;

    SplitMix64 rng(0xB51A1EULL);

    std::vector<double> x(n);
    std::vector<double> y(n);
    std::vector<double> y_true(n);
    for (std::size_t i = 0; i < n; ++i) {
        x[i] = xmin + (xmax - xmin) * static_cast<double>(i) / static_cast<double>(n - 1);
        y_true[i] = std::sin(20.0 / (x[i] + 0.25));
        y[i] = y_true[i] + noise_sigma * rng.normal();
    }

    PSplineOptions opts;
    opts.lambda = 0.5;
    opts.penalty_order = 2;
    const auto fit = fit_pspline(x, y, 3, 300, opts);

    std::vector<double> grid_x(200);
    for (std::size_t i = 0; i < grid_x.size(); ++i)
        grid_x[i] = xmin + (xmax - xmin) * static_cast<double>(i)
                          / static_cast<double>(grid_x.size() - 1);

    const auto grid_y = fit.predict(grid_x);
    std::vector<double> grid_true(grid_x.size());
    for (std::size_t i = 0; i < grid_x.size(); ++i)
        grid_true[i] = std::sin(20.0 / (grid_x[i] + 0.25));

    auto plot = RPlot::create();
    plot.points(x, y, "samples", {37, 99, 235}, 6.0)
        .line(grid_x,
              grid_true,
              "true signal",
              {22, 163, 74},
              2.0)
        .line(grid_x,
              grid_y,
              "cubic B-spline fit",
              {220, 38, 38},
              3.0)
        .title("Nonlinear Regression With Cubic B-Splines")
        .x_label("x")
        .y_label("y")
        .background({250, 250, 252});

    try {
        plot.show("datamunge-plot-ex");
        std::cout << "viewer = shown\n";
    } catch (const std::exception& ex) {
        std::cout << "viewer = skipped (" << ex.what() << ")\n";
    }

    plot.save_svg("build/org/bspline_regression.svg");

    std::cout << std::fixed << std::setprecision(4)
              << "n = " << n << "\n"
              << "noise sigma = " << noise_sigma << "\n"
              << "basis functions = " << fit.basis.basis_size() << "\n"
              << "lambda = " << fit.lambda << "\n"
              << "effective dof = " << fit.effective_degrees_of_freedom << "\n"
              << "R^2 = " << fit.r_squared << "\n"
              << "adj R^2 = " << fit.adjusted_r_squared << "\n";
    return 0;
}
