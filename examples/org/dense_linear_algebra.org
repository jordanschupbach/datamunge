This example compiles and runs a small C++ program against the local
=datamunge= headers. The batch exporter in [[file:../../init.el][init.el]]
loads the repo's =direnv= environment first, so the compiler and other project
tools are available to the source block.

#+begin_src shell
mkdir -p build/org
cat > build/org/dense_linear_algebra.cpp <<'CPP'
#include <datamunge/linalg/linalg.hpp>

#include <iostream>

using datamunge::linalg::DenseMatrix;

int main() {
    DenseMatrix<double> a(2, 2, {1, 2, 3, 4});
    DenseMatrix<double> b(2, 2, {5, 6, 7, 8});

    auto sum = a + b;
    auto product = a * b;

    std::cout << "sum(0,0) = " << sum(0, 0) << "\n";
    std::cout << "sum(1,1) = " << sum(1, 1) << "\n";
    std::cout << "product(0,0) = " << product(0, 0) << "\n";
    std::cout << "product(1,1) = " << product(1, 1) << "\n";
    std::cout << "frobenius(a) = " << a.norm_frobenius() << "\n";
    return 0;
}
CPP

c++ -std=c++17 -Iinclude build/org/dense_linear_algebra.cpp \
  -o build/org/dense_linear_algebra

./build/org/dense_linear_algebra
#+end_src

This second block fits a nonlinear regression curve using a cubic B-spline basis.
The basis matrix is evaluated sparsely by default, the coefficients are fit with
=sparse_linear_regression=, and the resulting smooth curve is rendered to SVG.

#+begin_src shell
mkdir -p build/org
cat > build/org/bspline_regression.cpp <<'CPP'
#include <datamunge/fda/bspline.hpp>
#include <datamunge/linalg/csr.hpp>
#include <datamunge/linalg/regression.hpp>
#include <datamunge/plot/plot.hpp>

#include <iomanip>
#include <iostream>
#include <vector>

using datamunge::fda::BSplineBasis;
using datamunge::linalg::sparse_linear_regression;
using namespace datamunge::plot;

int main() {
    const std::vector<double> x = {0.00, 0.10, 0.22, 0.35, 0.48, 0.62, 0.76, 0.90,
                                   1.05, 1.20, 1.38, 1.55, 1.72, 1.88, 2.00};
    const std::vector<double> y = {0.18, 0.52, 0.91, 1.05, 0.82, 0.22, -0.41, -0.88,
                                   -0.97, -0.62, 0.03, 0.71, 1.02, 0.79, 0.28};

    const auto basis = BSplineBasis::open_uniform(3, 8, x.front(), x.back());
    const auto design_sparse = basis.evaluate(x);
    auto fit = sparse_linear_regression(design_sparse, y);

    std::vector<double> grid_x(200);
    const double xmin = x.front();
    const double xmax = x.back();
    for (std::size_t i = 0; i < grid_x.size(); ++i)
        grid_x[i] = xmin + (xmax - xmin) * static_cast<double>(i)
                          / static_cast<double>(grid_x.size() - 1);

    const auto grid_design = datamunge::linalg::to_csr(basis.evaluate(grid_x));
    const auto grid_y = grid_design.spmv(fit.coefficients);

    auto plot = ScatterPlot::create();
    plot.points(x, y, "samples", {37, 99, 235}, 6.0)
        .line(grid_x,
              grid_y,
              "cubic B-spline fit",
              {220, 38, 38},
              3.0)
        .title("Nonlinear Regression With Cubic B-Splines")
        .x_label("x")
        .y_label("y")
        .background({250, 250, 252});

    plot.save_svg("build/org/bspline_regression.svg");

    std::cout << std::fixed << std::setprecision(4)
              << "basis functions = " << basis.basis_size() << "\n"
              << "degrees of freedom = " << fit.degrees_of_freedom << "\n"
              << "R^2 = " << fit.r_squared << "\n"
              << "adj R^2 = " << fit.adjusted_r_squared << "\n";
    return 0;
}
CPP

c++ -std=c++17 -Iinclude \
  build/org/bspline_regression.cpp \
  src/datamunge/fda/bspline.cpp \
  src/datamunge/plot.cpp \
  -o build/org/bspline_regression

./build/org/bspline_regression
#+end_src
